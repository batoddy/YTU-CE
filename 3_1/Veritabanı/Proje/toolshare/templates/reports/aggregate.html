SQL Sorgusu
SELECT c.name, COUNT(t.id), AVG(t.daily_price), SUM(t.daily_price), MAX(t.daily_price), MIN(t.daily_price)
FROM categories c JOIN tools t ON c.id = t.category_id
GROUP BY c.id, c.name HAVING COUNT(t.id) >= 1
ORDER BY COUNT(t.id) DESC
